# petsim — the pet's WAMR on your Mac: same WAMR 2.4 source and the same config as the
# firmware (fast interpreter, builtin libc, ref types, no WASI/AOT), so an app that loads
# here loads on the pet, and one the pet would reject fails here first.
#   cmake -S sim -B sim/build && cmake --build sim/build      (build.sh does this once)
cmake_minimum_required(VERSION 3.16)
project(petsim C)
if(NOT DEFINED WAMR_ROOT)
  set(WAMR_ROOT ${CMAKE_CURRENT_LIST_DIR}/../../jc3248-pet-idf/managed_components/espressif__wasm-micro-runtime)
endif()
if(NOT EXISTS ${WAMR_ROOT}/build-scripts/runtime_lib.cmake)
  message(FATAL_ERROR "WAMR source not found at ${WAMR_ROOT}: build lab/jc3248-pet-idf once (it fetches espressif/wasm-micro-runtime)")
endif()
set(WAMR_BUILD_PLATFORM darwin)
if(CMAKE_SYSTEM_PROCESSOR MATCHES "arm64|aarch64")
  set(WAMR_BUILD_TARGET AARCH64)
else()
  set(WAMR_BUILD_TARGET X86_64)
endif()
set(WAMR_BUILD_INTERP 1)
set(WAMR_BUILD_FAST_INTERP 1)
set(WAMR_BUILD_AOT 0)
set(WAMR_BUILD_JIT 0)
set(WAMR_BUILD_LIBC_BUILTIN 1)
set(WAMR_BUILD_LIBC_WASI 0)
set(WAMR_BUILD_REF_TYPES 1)
set(WAMR_BUILD_LIB_PTHREAD 0)
include(${WAMR_ROOT}/build-scripts/runtime_lib.cmake)
add_library(vmlib ${WAMR_RUNTIME_LIB_SOURCE})
add_executable(petsim petsim.c)
target_link_libraries(petsim vmlib -lm -lpthread)
